\section*{Proof that $\sqrt{7}$ is irrational}

\subsection*{Permitted external facts}

We use the following standard facts without further proof.

\begin{enumerate}
    \item \textbf{The Division Algorithm:} for any integers $a$ and $d$ with $d>0$, there exist unique integers $q$ and $r$ with
    \[
        a = dq + r \quad\text{and}\quad 0 \le r < d.
    \]
    By uniqueness, $r = 0$ if and only if $d \mid a$, and $r \neq 0$ if and only if $d \nmid a$.

    \item \textbf{Euclid's Lemma:} if $p$ is prime and $p \mid ab$, then $p \mid a$ or $p \mid b$.

    \item \textbf{The Well-Ordering Principle:} every nonempty set of positive integers has a least element.

    \item \textbf{Closure:} the integers are closed under addition, subtraction, and multiplication.

    \item \textbf{Divisibility definition:} $d \mid a$ means there exists an integer $k$ such that $a = dk$. In particular, $1 \mid n$ for every integer $n$ (since $n = 1 \cdot n$), so $1$ is always a divisor.

    \item \textbf{Ordered-ring and ordered-field properties (freely used throughout):}
    \begin{itemize}
        \item \emph{Trichotomy:} every nonzero integer (or real number) is either positive or negative; in particular, if $b < 0$ then $-b > 0$.
        \item \emph{Integer discreteness:} every positive integer $n$ satisfies $n \ge 1$.
        \item \emph{Product of positives:} if $a > 0$ and $b > 0$ (integers, rationals, or reals), then $ab > 0$.
        \item \emph{Quotient of positives:} if $a > 0$ and $b > 0$ then $a/b > 0$.
        \item \emph{Order compatibility with multiplication:} if $a \ge b$ and $c > 0$, then $ac \ge bc$; if $a > b$ and $c > 0$, then $ac > bc$.
        \item \emph{Order compatibility with addition:} $x > y$ if and only if $x - y > 0$.
        \item \emph{Integer discreteness (lower bound):} if $n$ is an integer with $n > k$ for some integer $k$, then $n \ge k+1$. In particular, $n > 0$ implies $n \ge 1$, and $n > 1$ implies $n \ge 2$.
        \item \emph{Cancellation/division:} if $a = bc$ and $c \ne 0$, then $a/c = b$.
        \item \emph{Fraction cancellation:} if $c \ne 0$, then $(ac)/(bc) = a/b$.
        \item \emph{Negation preserves equality:} $(-a)/(-b) = a/b$ for $b \ne 0$.
        \item \emph{Squaring a fraction:} $(a/b)^2 = a^2/b^2$ for $b \ne 0$.
        \item \emph{Positive square root:} $\sqrt{x}$ denotes the unique positive real satisfying $(\sqrt{x})^2 = x$, for $x > 0$.
    \end{itemize}
\end{enumerate}

\begin{lemma}\label{lem:div}
For any integer $n$, if $7 \mid n^2$ then $7 \mid n$.
\end{lemma}

\begin{proof}
\textbf{Step 1. Primality of $7$.}

A positive integer $p$ is \emph{prime} if $p > 1$ and its only positive divisors are $1$ and $p$.
Since $7 = 6 + 1 > 1$, the condition $7 > 1$ is satisfied.
We verify that no integer $d$ with $2 \le d \le 6$ divides $7$, using the Division Algorithm (in each case the remainder $r$ satisfies $0 \le r < d$, confirmed by inspection):
\[
  7 = 2 \cdot 3 + 1, \quad r_2 = 1,\; 0 \le 1 < 2, \quad r_2 \ne 0, \quad\text{so } 2 \nmid 7;
\]
\[
  7 = 3 \cdot 2 + 1, \quad r_3 = 1,\; 0 \le 1 < 3, \quad r_3 \ne 0, \quad\text{so } 3 \nmid 7;
\]
\[
  7 = 4 \cdot 1 + 3, \quad r_4 = 3,\; 0 \le 3 < 4, \quad r_4 \ne 0, \quad\text{so } 4 \nmid 7;
\]
\[
  7 = 5 \cdot 1 + 2, \quad r_5 = 2,\; 0 \le 2 < 5, \quad r_5 \ne 0, \quad\text{so } 5 \nmid 7;
\]
\[
  7 = 6 \cdot 1 + 1, \quad r_6 = 1,\; 0 \le 1 < 6, \quad r_6 \ne 0, \quad\text{so } 6 \nmid 7.
\]
We now argue the case split is exhaustive.
Let $d > 1$ be any positive divisor of $7$; by the divisibility definition, $7 = dk$ for some integer $k$.
Since $7 > 0$ and $d > 0$, by the quotient-of-positives property $k = 7/d > 0$, so $k$ is a positive integer.
By integer discreteness, $k \ge 1$.
Therefore $d = 7/k \le 7/1 = 7$ (by the cancellation property and order compatibility with multiplication: $k \ge 1$ and $d > 0$ imply $dk \ge d \cdot 1 = d$, giving $d \le dk = 7$).
Thus every positive divisor $d > 1$ of $7$ satisfies $2 \le d \le 7$.
For $2 \le d \le 6$ we checked above that $d \nmid 7$.
For $d = 1$: $7 = 1 \cdot 7$ and $7 \in \mathbb{Z}$, so $1 \mid 7$ by the divisibility definition.
For $d = 7$: $7 = 7 \cdot 1$ and $1 \in \mathbb{Z}$, so $7 \mid 7$ by the divisibility definition.
Every positive divisor $d > 1$ of $7$ satisfies $2 \le d \le 7$: the lower bound $d \ge 2$ follows from integer discreteness applied to $d > 1$ (since $d$ is a positive integer with $d > 1$, integer discreteness gives $d \ge 2$); the upper bound $d \le 7$ was established above. Since $d$ is an integer with $2 \le d \le 7$, it belongs to $\{2,3,4,5,6,7\}$, and we have checked all cases.
Therefore the only positive divisors of $7$ are $1$ and $7$, confirming that $7$ is prime.

\textbf{Step 2. Apply Euclid's Lemma.}

Suppose $7 \mid n^2$.  Since the integers are closed under multiplication, $n^2 = n \cdot n$ is an integer.
Since $7$ is prime (Step~1) and $7 \mid n \cdot n$, Euclid's Lemma gives $7 \mid n$ or $7 \mid n$; in either case $7 \mid n$.
\end{proof}

\begin{theorem}\label{thm:irrational}
$\sqrt{7}$ is irrational.
\end{theorem}

\begin{proof}
\textbf{Step 1. Setup.}

Assume for contradiction that $\sqrt{7}$ is rational.
By definition there exist integers $a, b$ with $b \ne 0$ such that $\sqrt{7} = a/b$.
By trichotomy, either $b > 0$ or $b < 0$ (since $b \ne 0$).
If $b < 0$, replace $(a, b)$ by $(-a, -b)$: since $b < 0$, we have $-b > 0$ by trichotomy, and $(-a)/(-b) = a/b$ by the negation-preserves-equality property, so the equation $\sqrt{7} = a/b$ is preserved.
In either case, we may assume $b > 0$.
Since $\sqrt{7}$ is the positive square root of $7$ and $7 > 0$, we have $\sqrt{7} > 0$.
Multiplying both sides of $\sqrt{7} = a/b$ by $b > 0$ gives $a = b\sqrt{7}$.
Since $b > 0$ and $\sqrt{7} > 0$, the product $a = b\sqrt{7}$ is a product of two positive numbers, hence $a > 0$.

\textbf{Step 2. Well-ordering.}

Define
\[
  S = \{ q \in \mathbb{Z}_{>0} : q\sqrt{7} \in \mathbb{Z}_{>0} \}.
\]
Since $b \in \mathbb{Z}_{>0}$ and $b\sqrt{7} = a \in \mathbb{Z}_{>0}$, we have $b \in S$, so $S$ is nonempty.
By the Well-Ordering Principle, $S$ has a least element $q_0$.
Set $p_0 = q_0 \sqrt{7} \in \mathbb{Z}_{>0}$.
Then $p_0 > 0$ (since $q_0 > 0$ and $\sqrt{7} > 0$) and $\sqrt{7} = p_0/q_0$.

\textbf{Step 3. $7$ divides $p_0$.}

Squaring both sides of $\sqrt{7} = p_0/q_0$ gives $(\sqrt{7})^2 = (p_0/q_0)^2$, i.e., $7 = p_0^2/q_0^2$.
Since $q_0 \in \mathbb{Z}_{>0}$, the square $q_0^2 = q_0 \cdot q_0$ is a positive integer by closure.
Multiplying both sides by $q_0^2$ yields $p_0^2 = 7q_0^2$.
Hence $7 \mid p_0^2$.
By Lemma~\ref{lem:div} (with $n = p_0 \in \mathbb{Z}$), $7 \mid p_0$.
By the divisibility definition, there exists an integer $k$ such that $p_0 = 7k$.
Since $p_0 > 0$ and $7 > 0$, we have $k = p_0/7 > 0$, so $k \in \mathbb{Z}_{>0}$.

\textbf{Step 4. $7$ divides $q_0$.}

Substituting $p_0 = 7k$ into $p_0^2 = 7q_0^2$ gives $(7k)^2 = 7q_0^2$, i.e., $49k^2 = 7q_0^2$.
Since $7 \ne 0$, dividing both sides by $7$ gives $7k^2 = q_0^2$.
Since $k > 0$, we have $k^2 = k \cdot k > 0$, and $k^2$ is an integer by closure.
Hence $7 \mid q_0^2$.
By Lemma~\ref{lem:div} (with $n = q_0 \in \mathbb{Z}$), $7 \mid q_0$.
By the divisibility definition, there exists an integer $m$ such that $q_0 = 7m$.
Since $q_0 > 0$ and $7 > 0$, we have $m = q_0/7 > 0$, so $m \in \mathbb{Z}_{>0}$.

\textbf{Step 5. Contradiction.}

From $\sqrt{7} = p_0/q_0 = 7k/(7m)$, since $m > 0$ (hence $m \ne 0$) and $7 \ne 0$, we may cancel the common factor $7$ in numerator and denominator (by fraction cancellation) to obtain $\sqrt{7} = k/m$.
Multiplying both sides by $m > 0$ gives $m\sqrt{7} = k \in \mathbb{Z}_{>0}$, so $m \in S$.

It remains to show $m < q_0$, contradicting the minimality of $q_0$.
Since $q_0 = 7m$ and $7 \ge 2$ (by inspection: $7 = 2 + 5$), the order-compatibility-with-multiplication property gives $7m \ge 2m$ (multiplying $7 \ge 2$ by $m > 0$).
Also, $2m - m = m > 0$, and by order-compatibility-with-addition ($x - y > 0 \Leftrightarrow x > y$), this gives $2m > m$.
Therefore $q_0 = 7m \ge 2m > m$, i.e., $m < q_0$.
Thus $m \in S$ and $m < q_0$, contradicting the minimality of $q_0$.

Therefore the assumption that $\sqrt{7}$ is rational is false, and $\sqrt{7}$ is irrational.
\end{proof}
